\\[ W\_{\mathbb Q} =\operatorname{Hom}\_{\mathbb Q[G]} (U\_{\mathbb Q},H^1(C,\mathbb Q)) \\]

\\[ \det(qh)=q^8\det h =N\_{K/\mathbb Q}(q^4)\det h. \\]
